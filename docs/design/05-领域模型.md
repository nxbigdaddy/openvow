## 6.1 实体关系

```
project 1─n graph 1─n task n─n task（task_deps）
                  task 1─n submission
                  task 1─n task_files n─1 artifact
                  task 1─n checkpoint
                  task 1─n event
project 1─n terminal
integration 1─n integration_record
```

## 6.2 数据模型（PostgreSQL 规范版）

```sql
-- 项目与仓库镜像
CREATE TABLE projects (
  id            text PRIMARY KEY,
  name          text NOT NULL,
  repo_url      text NOT NULL,
  canonical_ref text NOT NULL DEFAULT 'main',
  head_commit   text,
  created_at    timestamptz NOT NULL DEFAULT now()
);

-- 终端（外部执行体）
CREATE TABLE terminals (
  id           text PRIMARY KEY,
  project_id   text REFERENCES projects(id),
  name         text NOT NULL,
  scope        text NOT NULL CHECK (scope IN ('planner','worker','admin')),
  token_hash   bytea NOT NULL,
  labels       text[] NOT NULL DEFAULT '{}',
  trust        numeric(4,3) NOT NULL DEFAULT 0,
  accepted_n   int NOT NULL DEFAULT 0,
  total_n      int NOT NULL DEFAULT 0,
  banned_at    timestamptz,
  last_seen_at timestamptz
);

-- 图（每次拆解一个版本，永不物理删除）
CREATE TABLE graphs (
  id              text PRIMARY KEY,
  project_id      text NOT NULL REFERENCES projects(id),
  version         int  NOT NULL,
  state           text NOT NULL CHECK (state IN ('draft','validated','approved','active','superseded','rejected')),
  requirement_ref text NOT NULL,
  base_commit     text NOT NULL,
  body            jsonb NOT NULL,          -- task_graph.json 原文
  sha256          text NOT NULL,
  supersedes      text REFERENCES graphs(id),
  approved_by     text,
  approved_at     timestamptz,
  created_at      timestamptz NOT NULL DEFAULT now(),
  UNIQUE (project_id, version)
);

-- 任务
CREATE TABLE tasks (
  id           text PRIMARY KEY,
  graph_id     text REFERENCES graphs(id),
  node_id      text,                        -- 图内节点 id；平台生成的为空
  kind         text NOT NULL CHECK (kind IN
                 ('explore','decompose','interface','test','implement','integration','repair')),
  title        text NOT NULL,
  writes       text[] NOT NULL,             -- 必须两两不相交
  reads        text[] NOT NULL DEFAULT '{}',
  provides     text[] NOT NULL DEFAULT '{}',
  consumes     text[] NOT NULL DEFAULT '{}',
  covers       text[] NOT NULL DEFAULT '{}',
  frozen       jsonb  NOT NULL DEFAULT '[]',-- [{"path","sha256"}]
  acceptance   jsonb  NOT NULL,
  requires     text[] NOT NULL DEFAULT '{}',
  state        text NOT NULL CHECK (state IN
                 ('queued','claimed','validating','accepted','rejected','merged','stale','needs_human')),
  attempts     int NOT NULL DEFAULT 0,
  max_attempts int NOT NULL DEFAULT 2,
  priority     int NOT NULL DEFAULT 100,
  lease_id     text,
  lease_until  timestamptz,
  terminal_id  text REFERENCES terminals(id),
  base_commit  text,
  merge_group  text,
  created_at   timestamptz NOT NULL DEFAULT now(),
  updated_at   timestamptz NOT NULL DEFAULT now()
);
CREATE INDEX tasks_claim_idx  ON tasks (state, priority DESC, created_at);
CREATE INDEX tasks_writes_idx ON tasks USING gin (writes);   -- 支撑 && 交叠判断

CREATE TABLE task_deps (
  parent_id text NOT NULL REFERENCES tasks(id),
  child_id  text NOT NULL REFERENCES tasks(id),
  PRIMARY KEY (parent_id, child_id)
);

-- 中间产物（断点续跑）
CREATE TABLE checkpoints (
  id         bigserial PRIMARY KEY,
  task_id    text NOT NULL REFERENCES tasks(id),
  attempt    int  NOT NULL,
  seq        int  NOT NULL,
  note       text,
  files      jsonb NOT NULL,
  created_at timestamptz NOT NULL DEFAULT now()
);

-- 内容寻址产物库
CREATE TABLE artifacts (
  sha256     text PRIMARY KEY,
  size       bigint NOT NULL,
  media_type text,
  created_at timestamptz NOT NULL DEFAULT now()
);

CREATE TABLE task_files (
  task_id text NOT NULL,
  attempt int  NOT NULL,
  path    text NOT NULL,
  sha256  text NOT NULL REFERENCES artifacts(sha256),
  PRIMARY KEY (task_id, attempt, path)
);

-- 提交记录（幂等键）
CREATE TABLE submissions (
  id         bigserial PRIMARY KEY,
  task_id    text NOT NULL,
  attempt    int  NOT NULL,
  lease_id   text NOT NULL,
  accepted   boolean NOT NULL,
  code       text,
  detail     jsonb,
  created_at timestamptz NOT NULL DEFAULT now(),
  UNIQUE (task_id, attempt)
);

-- 集成记录
CREATE TABLE integrations (
  id               bigserial PRIMARY KEY,
  merge_group      text NOT NULL,
  candidate_commit text,
  result           text NOT NULL CHECK (result IN ('merged','failed','timeout')),
  bundle           jsonb,
  created_at       timestamptz NOT NULL DEFAULT now()
);

-- 审计（只追加）
CREATE TABLE events (
  id      bigserial PRIMARY KEY,
  task_id text,
  kind    text NOT NULL,
  payload jsonb,
  at      timestamptz NOT NULL DEFAULT now()
);
```

**SQLite 差异（M0）**：`text[]` → JSON 文本；`jsonb` → `TEXT`；`timestamptz` → `TEXT`（RFC3339 UTC）；`bigserial` → `INTEGER PRIMARY KEY AUTOINCREMENT`；无 `&&` 运算与 GIN 索引（M0 不并行，不需要）。

## 6.3 任务状态机

```
queued ──claim──► claimed ──submit──► validating ──pass──► accepted ──► merged
  ▲                  │                    │ fail                  │
  └──lease 过期──────┘                    ▼                       │
  └──────────────────────────────  rejected（可重试）              │
                                      │ attempts 超限              │
                                      ▼                            │
                                  needs_human                      │
                                                                    │
  accepted ──图修订──► stale ──rebase──► queued                     │
```

规则：

- `claimed` 必须有未过期的 `lease_id`；
- `accepted` 是唯一的解锁条件（不是 `done`、不是终端自报 success）；
- `stale` 的产物禁止进入集成，但不删除；
- `needs_human` 是终态，只能由人重置。

## 6.4 图状态机

```
draft ──validate──► validated ──approve(人)──► active ──supersede──► superseded
  │                    │
  └── reject ◄─────────┘   → 回 planner 重拆（新 version）
```

规则：

- 只有 `active` 图下的任务可派发；
- 任何修订产生新 `version`，旧版 `superseded`，**永不物理删除**；
- `approve` 必须由 human token 调用，planner 无此权限。
